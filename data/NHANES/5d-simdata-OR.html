make_ids_unique <- function(ids) {
  dup_ids <- ids[duplicated(ids) | duplicated(ids, fromLast = TRUE)]
  unique_dup_ids <- unique(dup_ids)
  for (id in unique_dup_ids) {
    suffix <- 1
    indices <- which(ids == id)
    for (index in indices) {
      ids[index] <- paste0(ids[index], ".", suffix)
      suffix <- suffix + 1
    }
  }
  return(ids)
}

custom_simulate <- function(i, plasmode.data, analytic.data, demovars, lab.list, proxy.list.arg, nSim, intermediate_dir) { 
  df_for_join <- data.frame(
  id = plasmode.data[, i],
  EVENT = plasmode.data[, i + nSim],
  EXPOSURE = plasmode.data[, i + 2 * nSim]
  )
  
  plasmodeData.i <- plyr::join(x = df_for_join,
                               y = analytic.data, # This is full.data
                               by = "id",         
                               type = "left")

  if (nrow(plasmodeData.i) != nrow(df_for_join)) {
    warning(paste("Join for simulation", i, "resulted in different row count than expected (", nrow(plasmodeData.i), "vs", nrow(df_for_join),"). Check 'by' variable and data integrity."))
  }
  if (any(is.na(plasmodeData.i$id))) { # IDs should not become NA after a left join on x's key
    warning(paste("NA values found in 'id' column after join for iteration", i, ". This is unexpected. Review join logic."))
  }

  plasmodeData.i$outcome <- plasmodeData.i$EVENT 
  plasmodeData.i$exposure <- plasmodeData.i$EXPOSURE
  plasmodeData.i$EVENT <- NULL
  plasmodeData.i$EXPOSURE <- NULL
  plasmodeData.i$diabetes <- NULL
  plasmodeData.i$obese <- NULL
  plasmodeData.i$l1 <- NULL
  plasmodeData.i$l2 <- NULL
  plasmodeData.i$l3 <- NULL
  plasmodeData.i$l4 <- NULL
  plasmodeData.i$l5 <- NULL
  plasmodeData.i$l6 <- NULL
  
  plasmodeData.i$id <- make_ids_unique(plasmodeData.i$id) 
  
  saveRDS(plasmodeData.i, file = paste0(intermediate_dir, "/data_", i, ".rds"))
  return(plasmodeData.i)
}
